-- Project: Customer 360 and Marketing Attribution
-- Query 04: Marketing Revenue Analysis

SELECT
    traffic_source.source AS source,
    traffic_source.medium AS medium,
    COUNT(*) AS purchase_events,
    COUNT(DISTINCT user_pseudo_id) AS unique_purchasers,
    COUNT(DISTINCT ecommerce.transaction_id) AS total_transactions,
    ROUND(
        SUM(ecommerce.purchase_revenue_in_usd), 2
    ) AS total_revenue_usd,
    ROUND(
        AVG(ecommerce.purchase_revenue_in_usd), 2
    ) AS average_revenue_usd
FROM
    `bigquery-public-data.ga4_obfuscated_sample_ecommerce.events_*`
WHERE
    event_name = 'purchase'
    AND _TABLE_SUFFIX BETWEEN '20201101' AND '20210131'
GROUP BY
    source,
    medium
ORDER BY
    total_revenue_usd DESC;
